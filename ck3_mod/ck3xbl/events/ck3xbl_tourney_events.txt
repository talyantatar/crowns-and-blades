# Tournament bouts in Bannerlord (docs/PROTOCOL.md 29), seen by the player:
#   0001 (hidden): the option "Compete in Bannerlord yourself" of a contest's pivotal moment: the bout is logged and fought.
#   0002 "In the Lists": open while the bout is fought in Bannerlord; hosts the keypad widget.
#   0004 (hidden): the delivered winner decides the bout (or the melee) in CK3.
#   0005 (hidden): the day after a versus bout, both contestants' scores come back.
# root = the player. The chain carries scope:activity, scope:versus_contestant and scope:ck3xbl_d_tourney.

namespace = ck3xbl_tourney

ck3xbl_tourney.0001 = {
	type = character_event
	hidden = yes
	immediate = { ck3xbl_tourney_begin_effect = yes }
}

ck3xbl_tourney.0002 = {
	type = character_event
	title = ck3xbl_tourney.0002.t
	desc = ck3xbl_tourney.0002.desc
	theme = tournament_contest
	left_portrait = {
		character = root
		animation = aggressive_sword
	}
	right_portrait = {
		character = scope:ck3xbl_enemy_leader
		animation = aggressive_sword
	}
	widget = {
		gui = "ck3xbl_keypad"
		container = "custom_widgets_container"
	}

	option = {	# Always clickable (as battle event 0002): before word arrives it simply keeps waiting.
		name = ck3xbl_tourney.0002.a
		hidden_effect = { trigger_event = ck3xbl_tourney.0004 }
	}
	option = {	# The bridge or Bannerlord failed: the bout is decided as CK3 would have decided it.
		name = ck3xbl_tourney.0002.b
		trigger = {
			NOT = { ck3xbl_result_applied_trigger = yes }
		}
		hidden_effect = { ck3xbl_tourney_to_ck3_effect = yes }
	}
}

ck3xbl_tourney.0004 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { ck3xbl_result_ready_trigger = yes }
			ck3xbl_tourney_apply_effect = yes
		}
		else_if = {
			limit = {
				is_alive = yes
				NOT = { ck3xbl_result_applied_trigger = yes }
			}
			trigger_event = ck3xbl_tourney.0002
		}
	}
}

ck3xbl_tourney.0005 = {
	type = character_event
	hidden = yes
	immediate = {
		ck3xbl_tourney_restore_score_effect = yes
		scope:versus_contestant ?= { ck3xbl_tourney_restore_score_effect = yes }
	}
}
