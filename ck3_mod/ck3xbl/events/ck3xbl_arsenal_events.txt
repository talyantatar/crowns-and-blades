# The Arsenal (docs/PROTOCOL.md 51), seen by the player:
#   0001 (hidden): the decision opens it: the visit is logged and taken to Bannerlord (ck3xbl_arsenal_begin_effect).
#   0002 "The Arsenal": open while the player picks his gear in Bannerlord; hosts the keypad widget.
#   0004 (hidden): the visit is closed (ck3xbl_arsenal_apply_effect).

namespace = ck3xbl_arsenal

ck3xbl_arsenal.0001 = {
	type = character_event
	hidden = yes
	immediate = { ck3xbl_arsenal_begin_effect = yes }
}

ck3xbl_arsenal.0002 = {
	type = character_event
	title = ck3xbl_arsenal.0002.t
	desc = ck3xbl_arsenal.0002.desc
	theme = martial
	left_portrait = {
		character = root
		animation = personality_bold
	}
	widget = {
		gui = "ck3xbl_keypad"
		container = "custom_widgets_container"
	}

	option = {	# Always clickable (as battle event 0002): before word arrives it simply keeps waiting.
		name = ck3xbl_arsenal.0002.a
		hidden_effect = { trigger_event = ck3xbl_arsenal.0004 }
	}
	option = {	# The bridge or Bannerlord failed: nothing to keep.
		name = ck3xbl_arsenal.0002.b
		trigger = {
			NOT = { ck3xbl_result_applied_trigger = yes }
		}
		hidden_effect = { ck3xbl_abandon_battle_effect = yes }
	}
}

ck3xbl_arsenal.0004 = {
	type = character_event
	hidden = yes
	immediate = {
		if = {
			limit = { ck3xbl_result_ready_trigger = yes }
			ck3xbl_arsenal_apply_effect = yes
		}
		else_if = {
			limit = {
				is_alive = yes
				NOT = { ck3xbl_result_applied_trigger = yes }
			}
			trigger_event = ck3xbl_arsenal.0002
		}
	}
}
