# Helpers of the scenes for recording (events/ck3xbl_debug_events.txt, debug mode only).

# root = the player. scope:ck3xbl_dbg_foe = a lowborn fighter where the player is: a pool character, else a new warrior (as
# vanilla's travel dangers find their bandit).
ck3xbl_debug_lowborn_foe_effect = {
	random_pool_character = {
		province = root.location
		limit = {
			is_available_ai_adult = yes
			is_lowborn = yes
		}
		save_scope_as = ck3xbl_dbg_foe
		add_character_flag = wear_armor
	}
	if = {
		limit = { NOT = { exists = scope:ck3xbl_dbg_foe } }
		create_character = {
			template = new_warrior_character
			gender_female_chance = 0
			location = root.location
			culture = root.location.culture
			faith = root.location.faith
			save_scope_as = ck3xbl_dbg_foe
		}
		scope:ck3xbl_dbg_foe = { add_character_flag = wear_armor }
	}
}
